Iteration 7: Error exception

    import sys

    try:
        print("hello, my name is", sys.argv[1])
    except IndexError:
        print("Too few arguments")


Iteration 8: Refined user advise for error handling

    import sys

    if len(sys.argv) <2:
        print("Too few arguments")              
    elif len(sys.argv) > 2:
        print("Too many argiments")
    else:
        print("hello, my name is", sys.argv[1])     


Iteration 9: Good practice - dont hide the code you care about in an else statement.
                sys.exit(<"message">) - exit the program earlier 

    import sys

    if len(sys.argv) <2:
        sys.exit("Too few arguments")              
    elif len(sys.argv) > 2:
        sys.exit("Too many argiments")

    print("hello, my name is", sys.argv[1]) 



Iteration 10: multiple argument input
                slice - sys.argv[a:b] 

    import sys
    
    if len(sys.argv) <2:
        sys.exit("Too few arguments")    

    for arg in sys.argv[1:]:
        print("hello, my name is", arg)

Iteration 13: package management - 
                json (comes with python, no need to install with pip)
    import json
    import requests
    import sys

    if len(sys.argv) != 2:
        sys.exit()
    
    response = request.get("https://itunes.apple.com/search?entity=song&limit=1&term=" + sys.argv[1])

Iteration 14:  print a specific object from the json response
    import json
    import requests
    import sys

    if len(sys.argv) != 2:
        sys.exit()
    
    response = request.get("https://itunes.apple.com/search?entity=song&limit=10&term=" + sys.argv[1])

    o = response.json()
    for result in o["results"]:
        print(result["trackName"])
