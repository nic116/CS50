###################################################################################################
Iteration 1: from calculator import square   #import the function to test

def main():
    test_square()

def test_square():      #name convention test_<what i want to test>
    if square(2) != 4:
        print("2 square was not 4")
    if square(3) != 9:
        print("3 square was not 9")

if __name__ == "__main()__":
    main()                          #it will not print anything if it works


###################################################################################################
Iteration 2: assert - 
            we assert that a statement is true. If it is, nothing will happen. However, 
            if it isn't, error will pop up


def main():
    test_square()

def test_square():      
    assert square(2) = 4
    assert square(3) = 9

if __name__ == "__main()__":
    main() 


###################################################################################################
Iteration 3: try/except + assert
            long code but more consistent with the usual try/except concepts
            all the places that fail will show on terminal
def main():
    test_square()

def test_square():      
    try:
        assert square(2) = 4
    except AssertionError:
        print("2 square was not 4")
    try:
        assert square(3) = 9
    except AssertionError:
        print("3 square was not 9")
    try:
        assert square(-2) = 4
    except AssertionError:
        print("-2 square was not 4")
    try:
        assert square(-3) = 9
    except AssertionError:
        print("-3 square was not 9")
    try:
        assert square(0) = 0
    except AssertionError:
        print("0 square was not 0")

if __name__ == "__main()__":
    main()
